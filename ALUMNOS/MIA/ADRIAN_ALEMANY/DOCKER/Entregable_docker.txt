###SUM.PY:

import sys

# Comprobar si se pasaron dos argumentos
if len(sys.argv) != 3:
    print("Uso: python sum.py <num1> <num2>")
    sys.exit(1)

# Convertir los argumentos a números
try:
    num1 = float(sys.argv[1])
    num2 = float(sys.argv[2])
except ValueError:
    print("Error: ambos parámetros deben ser números.")
    sys.exit(1)

# Calcular la suma
resultado = num1 + num2
print(f"Sum: {resultado}")
